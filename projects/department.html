PHP & MySQL Department Portal Source

Backend scripts for session validation, student registration, and database records.

1. phpMyAdmin Database Schema (`dept_portal.sql`)
CREATE DATABASE IF NOT EXISTS pondi_dept_db;
USE pondi_dept_db;

CREATE TABLE IF NOT EXISTS students (
    reg_id VARCHAR(30) PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    course VARCHAR(50) NOT NULL,
    semester VARCHAR(10) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    registered_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
2. Registration Processor (`register.php`)
<?php
require_once 'db.php';

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $reg_id = trim($_POST['reg_id']);
    $name = trim($_POST['full_name']);
    $course = trim($_POST['course']);
    $sem = trim($_POST['semester']);
    $email = trim($_POST['email']);
    $password = password_hash($_POST['password'], PASSWORD_BCRYPT);

    $stmt = $conn->prepare("INSERT INTO students (reg_id, full_name, course, semester, email, password_hash) VALUES (?, ?, ?, ?, ?, ?)");
    $stmt->bind_param("ssssss", $reg_id, $name, $course, $sem, $email, $password);

    if ($stmt->execute()) {
        header("Location: login.php?registered=1");
    } else {
        echo "Error: " . $stmt->error;
    }
}
?>
3. Authentication & Session (`login.php`)
<?php
session_start();
require_once 'db.php';

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $id = trim($_POST['login_id']);
    $pass = $_POST['password'];

    $stmt = $conn->prepare("SELECT reg_id, full_name, password_hash FROM students WHERE reg_id = ? OR email = ?");
    $stmt->bind_param("ss", $id, $id);
    $stmt->execute();
    $result = $stmt->get_result();

    if ($user = $result->fetch_assoc()) {
        if (password_verify($pass, $user['password_hash'])) {
            $_SESSION['user_id'] = $user['reg_id'];
            $_SESSION['user_name'] = $user['full_name'];
            header("Location: dashboard.php");
            exit;
        }
    }
    $error = "Invalid Register ID or Password";
}
?>
