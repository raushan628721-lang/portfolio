PHP & MySQL Conditional Authentication Scripts

Complete backend code demonstrating role-based redirect and SQL fetching logic for CSCA 514 lab records.

1. MySQL Database Schema (`company_rbac.sql`)

CREATE DATABASE IF NOT EXISTS company_rbac;
USE company_rbac;

-- Users table storing credentials and roles
CREATE TABLE IF NOT EXISTS users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('employee', 'admin') NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Admin details table (viewed by authenticated employees)
CREATE TABLE IF NOT EXISTS admin_details (
    admin_id VARCHAR(20) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    designation VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL,
    contact_no VARCHAR(20) NOT NULL,
    office_room VARCHAR(50) NOT NULL
);

-- Employee details table (viewed by authenticated admins)
CREATE TABLE IF NOT EXISTS employee_details (
    emp_id VARCHAR(20) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    department VARCHAR(50) NOT NULL,
    role VARCHAR(50) NOT NULL,
    salary DECIMAL(10,2) NOT NULL,
    status VARCHAR(20) NOT NULL
);

2. PHP Authentication & Conditional Query (`authenticate.php`)

<?php
session_start();
require_once 'db.php';

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $username = trim($_POST['username']);
    $password = $_POST['password'];

    $stmt = $conn->prepare("SELECT id, username, full_name, password_hash, role FROM users WHERE username = ?");
    $stmt->bind_param("s", $username);
    $stmt->execute();
    $result = $stmt->get_result();

    if ($user = $result->fetch_assoc()) {
        if (password_verify($password, $user['password_hash'])) {
            $_SESSION['user_id'] = $user['id'];
            $_SESSION['username'] = $user['username'];
            $_SESSION['full_name'] = $user['full_name'];
            $_SESSION['role'] = $user['role'];

            // Conditional Routing based on lab criteria:
            if ($user['role'] === 'employee') {
                // If employee login, fetch and view all admin details
                header("Location: view_admins.php");
                exit;
            } else if ($user['role'] === 'admin') {
                // If admin login, fetch and view all employee details
                header("Location: view_employees.php");
                exit;
            }
        }
    }
    echo "<script>alert('Invalid credentials!'); window.location.href='login.php';</script>";
}
?>
