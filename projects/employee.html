PHP Backend & phpMyAdmin Database Implementation

Below are the official laboratory source codes for CSCA 514: MySQL database definition, database connection handle, and PHP insertion script.

1. phpMyAdmin SQL Table Schema (`database: company_db`) 
CREATE DATABASE IF NOT EXISTS company_db;
USE company_db;

CREATE TABLE IF NOT EXISTS employees (
    emp_id VARCHAR(20) PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    department VARCHAR(50) NOT NULL,
    designation VARCHAR(50) NOT NULL,
    salary DECIMAL(10, 2) NOT NULL,
    status VARCHAR(20) DEFAULT 'Active',
    joining_date DATE NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
2. Database Connection Script (`db_connect.php`) 
<?php
$servername = "localhost";
$username = "root";
$password = "";
$dbname = "company_db";

// Create MySQLi connection
$conn = new mysqli($servername, $username, $password, $dbname);

// Check connection
if ($conn->connect_error) {
    die("Database Connection failed: " . $conn->connect_error);
}
?>
3. Form Submission Handler (`insert_employee.php`) 
<?php
require_once 'db_connect.php';

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $emp_id = htmlspecialchars(trim($_POST['emp_id']));
    $name = htmlspecialchars(trim($_POST['name']));
    $email = htmlspecialchars(trim($_POST['email']));
    $department = htmlspecialchars(trim($_POST['department']));
    $designation = htmlspecialchars(trim($_POST['designation']));
    $salary = floatval($_POST['salary']);
    $status = htmlspecialchars(trim($_POST['status']));
    $joining_date = $_POST['joining_date'];

    // Prepared statement to prevent SQL Injection
    $stmt = $conn->prepare("INSERT INTO employees (emp_id, name, email, department, designation, salary, status, joining_date) VALUES (?, ?, ?, ?, ?, ?, ?, ?)");
    $stmt->bind_param("sssssdss", $emp_id, $name, $email, $department, $designation, $salary, $status, $joining_date);

    if ($stmt->execute()) {
        echo "<script>alert('Employee record successfully stored in database!'); window.location.href='employee.php';</script>";
    } else {
        echo "Error: " . $stmt->error;
    }

    $stmt->close();
    $conn->close();
}
?>
