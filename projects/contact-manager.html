Ex 8: Contact Management (CRUD)

PHP & MySQL CRUD Implementation

Complete lab code for CSCA 514: Database table structure and PHP CRUD API operations.

1. MySQL Database Schema (`contacts_db.sql`)

CREATE DATABASE IF NOT EXISTS contacts_db;
USE contacts_db;

CREATE TABLE IF NOT EXISTS contacts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    email VARCHAR(100) NOT NULL,
    category ENUM('Work', 'College', 'Personal') DEFAULT 'Personal',
    address TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

2. PHP Insert / Update Operations (`manage_contact.php`)

<?php
require_once 'db.php';

// Add New Contact
if (isset($_POST['action']) && $_POST['action'] === 'add') {
    $name = htmlspecialchars($_POST['name']);
    $phone = htmlspecialchars($_POST['phone']);
    $email = htmlspecialchars($_POST['email']);
    $category = htmlspecialchars($_POST['category']);
    $address = htmlspecialchars($_POST['address']);

    $stmt = $conn->prepare("INSERT INTO contacts (name, phone, email, category, address) VALUES (?, ?, ?, ?, ?)");
    $stmt->bind_param("sssss", $name, $phone, $email, $category, $address);
    $stmt->execute();
}

// Delete Contact
if (isset($_GET['delete_id'])) {
    $id = intval($_GET['delete_id']);
    $stmt = $conn->prepare("DELETE FROM contacts WHERE id = ?");
    $stmt->bind_param("i", $id);
    $stmt->execute();
}
?>
